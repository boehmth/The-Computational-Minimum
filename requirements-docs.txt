# MkDocs mit Material-Theme (das Buch-Layout)

# PyMdown-Erweiterungen fuer Admonitions, Code-Highlighting, Tabs, Math etc.
# (in mkdocs-material bereits transient enthalten, hier zur Sicherheit)
pymdown-extensions>=10.7

# Emoji-Twemoji-Support ist bereits in mkdocs-material enthalten;

# System-Bibliotheken (libcairo). Fuer eine reine Text-Site nicht noetig.

# SEO: minifiziert HTML/CSS/JS fuer schnellere Ladezeiten (Google-Ranking-Faktor)
mkdocs-minify-plugin>=0.8.0
